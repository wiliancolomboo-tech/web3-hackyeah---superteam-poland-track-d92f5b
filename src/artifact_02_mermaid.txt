flowchart TD
    %% Roles
    Client["Enterprise Client (Payer)"]
    Contractor["Contractor / Dev Shop (Payee)"]
    Arbiter["Designated Arbiter / Legal DAO"]
    Sentinel["SolVanguard Sentinel Worker"]
    Resend["Resend API Gateway"]
    Slack["Enterprise Webhook (Slack/Discord)"]

    %% On-Chain Subsystem
    subgraph Solana_Cluster["Solana Blockchain (Mainnet-Beta / Devnet)"]
        Program["SolVanguard Anchor Program"]
        EscrowPDA["Escrow State PDA [b'escrow', client, seed]"]
        VaultPDA["Token/SOL Vault PDA [b'vault', escrow]"]
        EventBus["On-Chain Program Event Bus"]
    end

    %% Lifecycle Steps
    Client -->|"1. initialize_escrow(milestones, arbiter, deadline)"| Program
    Program -->|"Create & Init"| EscrowPDA
    Client -->|"2. deposit_funds(amount)"| Program
    Program -->|"Transfer Lamports/SPL"| VaultPDA

    Contractor -->|"3. submit_milestone_proof(milestone_idx)"| Program
    
    alt Happy Path: Approval
        Client -->|"4a. release_milestone(milestone_idx)"| Program
        Program -->|"Transfer Milestone Share"| Contractor
        Program -->|"Emit MilestoneReleasedEvent"| EventBus
    else Dispute Path: Arbitration
        Contractor -->|"4b. dispute_milestone(milestone_idx)"| Program
        EscrowPDA -->|"State -> DISPUTED"| EscrowPDA
        Program -->|"Emit DisputeOpenedEvent"| EventBus
        Arbiter -->|"5. resolve_dispute(split_ratio_bps)"| Program
        Program -->|"Disburse Arbiter Split"| VaultPDA
        VaultPDA -->|"Disbursed to Client & Contractor"| Program
        Program -->|"Emit DisputeResolvedEvent"| EventBus
    end

    %% Sentinel & Observability
    EventBus -.->|"WebSocket logSubscribe / commitment: finalized"| Sentinel
    Sentinel -->|"Validate Base64 Event Signature"| Sentinel
    Sentinel -->|"Dispatch Audit Receipt (TLS 1.3)"| Resend
    Sentinel -->|"Dispatch Operational Alert"| Slack
    Resend -->|"Transactional PDF/Email"| Client
    Resend -->|"Receipt Confirmation"| Contractor
